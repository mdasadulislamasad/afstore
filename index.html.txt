  <script>
    // গুরুত্বপূর্ণ: আপনার Apps Script Web App URL টি এখানে বসান
    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxhzBDTXLDZ3ZFGPcseEeJ-lCjb7dCu72aXSlwlzk8VnyZLT1sQttKc5-jOFZqfUY6EAg/exec";

    const unitPrice = 1250;
    const deliveryCharge = 80;
    let quantity = 1;

    function updateQty(change) {
      if (quantity + change >= 1) {
        quantity += change;
        document.getElementById('qty-text').innerText = quantity;
        
        const subtotal = quantity * unitPrice;
        document.getElementById('subtotal').innerText = subtotal;
        document.getElementById('grand-total').innerText = subtotal + deliveryCharge;
      }
    }

    document.getElementById('orderForm').addEventListener('submit', function(e) {
      e.preventDefault();
      
      const submitBtn = document.getElementById('submitBtn');
      submitBtn.disabled = true;
      submitBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> অর্ডার প্রসেস হচ্ছে...';

      const orderData = {
        name: document.getElementById('custName').value,
        phone: document.getElementById('custPhone').value,
        address: document.getElementById('custAddress').value,
        product: "Ultra Smart Watch T800",
        quantity: quantity,
        totalPrice: (quantity * unitPrice) + deliveryCharge
      };

      fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(orderData)
      })
      .then(() => {
        document.getElementById('orderForm').classList.add('hidden');
        document.getElementById('successMsg').classList.remove('hidden');
      })
      .catch(error => {
        alert("অর্ডার পাঠাতে সমস্যা হয়েছে, দয়া করে আবার চেষ্টা করুন।");
        submitBtn.disabled = false;
        submitBtn.innerText = "অর্ডার কনফার্ম করুন (ক্যাশ অন ডেলিভারি)";
      });
    });
  </script>
